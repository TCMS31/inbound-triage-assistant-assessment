# Real request/response pairs, captured from a curl container on the compose
# network (docker compose up --build, LLM_PROVIDER=auto, NO API key set).
# From the host, the same calls go to http://localhost:8762.

$ curl -s localhost:8762/api/health
{"status":"ok"}

$ curl -s localhost:8762/api/inbound | jq length
13

# A message that does need a model, with no key configured. HTTP 200:

$ curl -s -i -X POST localhost:8762/api/triage/inb-001 | head -1
HTTP/1.1 200 OK

$ curl -s -X POST localhost:8762/api/triage/inb-001
{
  "id": "inb-001",
  "result": null,
  "source": "llm",
  "flags": [],
  "error": true,
  "error_message": "No LLM API key configured — set ANTHROPIC_API_KEY or OPENAI_API_KEY in backend/.env, or set LLM_PROVIDER=stub to run offline with the rule-based baseline."
}

# An unknown id is a 404, and so is an encoded path-traversal attempt:

$ curl -s -X POST localhost:8762/api/triage/inb-999
{"detail":"no inbound message with id 'inb-999'"}

$ curl -s -o /dev/null -w '%{http_code}' --path-as-is -X POST localhost:8762/api/triage/..%2F..%2Fetc%2Fpasswd
404
